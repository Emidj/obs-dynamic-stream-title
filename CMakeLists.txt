cmake_minimum_required(VERSION 3.28)
project(obs-dynamic-stream-title VERSION 1.0.0 LANGUAGES CXX RC)
# Keep the binary on the oldest Qt ABI used for our OBS 32.x baseline.
# Building with a newer Qt can add symbols missing from older OBS installs.
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This release targets Windows x64.")
endif()
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
qt_add_resources(EMOJI_RESOURCE data/emoji.qrc)
add_library(dst-ui STATIC src/dock.cpp src/emoji-picker.cpp src/ui-language.cpp ${EMOJI_RESOURCE})
target_include_directories(dst-ui PUBLIC src shared)
target_compile_features(dst-ui PUBLIC cxx_std_17)
target_compile_definitions(dst-ui PUBLIC DST_VERSION="${PROJECT_VERSION}")
target_link_libraries(dst-ui PUBLIC Qt6::Widgets)
add_library(dst-twitch-client STATIC src/twitch-core-client.cpp)
target_include_directories(dst-twitch-client PUBLIC src shared)
target_compile_features(dst-twitch-client PUBLIC cxx_std_17)
add_library(obs-dynamic-stream-title MODULE src/plugin.cpp)
target_link_libraries(obs-dynamic-stream-title PRIVATE dst-ui dst-twitch-client OBS::libobs OBS::obs-frontend-api)
include(CTest)
if(BUILD_TESTING)
  add_executable(dst-language-tests tests/language-tests.cpp)
  target_link_libraries(dst-language-tests PRIVATE dst-ui)
  add_test(NAME five-language-ui COMMAND dst-language-tests)
  set_tests_properties(five-language-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal;QT_PLUGIN_PATH=$<TARGET_FILE_DIR:Qt6::Core>/../plugins" ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>" TIMEOUT 20)
  add_executable(dst-language-preview WIN32 EXCLUDE_FROM_ALL tests/language-preview.cpp)
  target_link_libraries(dst-language-preview PRIVATE dst-ui)
  add_executable(dst-automation-stress EXCLUDE_FROM_ALL tests/automation-stress.cpp)
  target_link_libraries(dst-automation-stress PRIVATE dst-ui)
  add_executable(dst-composer-tests tests/composer-tests.cpp)
  target_include_directories(dst-composer-tests PRIVATE src)
  target_compile_features(dst-composer-tests PRIVATE cxx_std_17)
  target_link_libraries(dst-composer-tests PRIVATE Qt6::Core)
  add_test(NAME title-composer-isolation COMMAND dst-composer-tests)
  set_tests_properties(title-composer-isolation PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
  add_executable(dst-composer-preview EXCLUDE_FROM_ALL tests/composer-preview.cpp)
  target_link_libraries(dst-composer-preview PRIVATE dst-ui)
  add_executable(dst-compatibility-tests tests/compatibility-tests.cpp)
  target_include_directories(dst-compatibility-tests PRIVATE src)
  target_compile_features(dst-compatibility-tests PRIVATE cxx_std_17)
  add_test(NAME obs-version-range COMMAND dst-compatibility-tests)
  add_executable(dst-tests tests/dock-tests.cpp)
  target_link_libraries(dst-tests PRIVATE dst-ui)
  add_test(NAME dock-state-separation COMMAND dst-tests)
  set_tests_properties(dock-state-separation PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=minimal;QT_PLUGIN_PATH=$<TARGET_FILE_DIR:Qt6::Core>/../plugins"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>" TIMEOUT 20)
endif()
install(TARGETS obs-dynamic-stream-title LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
option(DST_BUILD_OVERLAY_INTEGRATION "Build isolated Overlay integration and its regression suites" OFF)
if(TRUE)
 find_package(CURL REQUIRED CONFIG)
 find_package(Qt6 REQUIRED COMPONENTS Network)
 if(DST_BUILD_OVERLAY_INTEGRATION)
  add_subdirectory(integration/overlay)
 else()
  add_subdirectory(integration/overlay EXCLUDE_FROM_ALL)
 endif()
 target_link_libraries(obs-dynamic-stream-title PRIVATE oto-shared-provider)
 install(TARGETS oto-shared-provider RUNTIME DESTINATION bin/64bit)
 if(BUILD_TESTING)
  add_executable(dst-shared-core-tests tests/shared-core-tests.cpp)
  target_link_libraries(dst-shared-core-tests PRIVATE oto-shared-provider dst-twitch-client)
  add_test(NAME shared-twitch-abi-lifecycle COMMAND dst-shared-core-tests)
  set_tests_properties(shared-twitch-abi-lifecycle PROPERTIES TIMEOUT 20
   ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:CURL::libcurl_shared>")
 endif()
 if(BUILD_TESTING)
  set_property(TEST shared-twitch-abi-lifecycle APPEND PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:OBS::libobs>;PATH=path_list_prepend:$<TARGET_FILE_DIR:oto-shared-provider>")
 endif()
endif()


# One independent release contract, embedded as inert PE resources in all consumers.
file(READ "${CMAKE_SOURCE_DIR}/shared/twitch-core-version.json" NEXUS_CORE_JSON)
string(JSON NEXUS_CORE_VERSION GET "${NEXUS_CORE_JSON}" Version)
string(JSON NEXUS_CORE_ABI GET "${NEXUS_CORE_JSON}" Abi)
string(REPLACE "." "," NEXUS_CORE_RC_VERSION "${NEXUS_CORE_VERSION}")
string(APPEND NEXUS_CORE_RC_VERSION ",0")
foreach(consumer TITLE OVERLAY)
 string(TOLOWER "${consumer}" consumer_file)
 file(READ "${CMAKE_SOURCE_DIR}/shared/contracts/${consumer_file}.json" consumer_json)
 string(JSON NEXUS_${consumer}_MIN GET "${consumer_json}" MinVersion)
 string(JSON NEXUS_${consumer}_MAX GET "${consumer_json}" MaxVersionExclusive)
 string(JSON NEXUS_${consumer}_ABI GET "${consumer_json}" Abi)
endforeach()
configure_file(shared/twitch-core-version.h.in generated/twitch-core-version.h @ONLY)
target_include_directories(oto-shared-provider PUBLIC "${CMAKE_BINARY_DIR}/generated")
set(NEXUS_CONTRACT_FILE "${CMAKE_SOURCE_DIR}/shared/twitch-core-version.json")
configure_file(shared/twitch-core.rc.in generated/twitch-core.rc @ONLY)
target_sources(oto-shared-provider PRIVATE "${CMAKE_BINARY_DIR}/generated/twitch-core.rc")
foreach(pair "obs-dynamic-stream-title|title" "obs-twitch-overlays|overlay")
 string(REPLACE "|" ";" parts "${pair}")
 list(GET parts 0 target)
 list(GET parts 1 contract)
 set(NEXUS_CONTRACT_FILE "${CMAKE_SOURCE_DIR}/shared/contracts/${contract}.json")
 configure_file(shared/twitch-contract.rc.in "generated/${contract}-contract.rc" @ONLY)
 target_sources(${target} PRIVATE "${CMAKE_BINARY_DIR}/generated/${contract}-contract.rc")
endforeach()
